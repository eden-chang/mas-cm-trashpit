# 5.3 - 알림 시스템

## 개요

마스토돈을 통해 플레이어에게 알림을 전송합니다.

---

## 알림 유형

| 유형 | 트리거 | 예시 |
|------|--------|------|
| 아이템 수신 | 양도 받았을 때 | "발트님이 로프를 양도했습니다!" |
| 공간 부족 | 획득 시 가방 부족 | "가방 공간이 부족합니다!" |
| 자동 삭제 경고 | 삭제 1시간 전 | "주변 아이템이 곧 삭제됩니다!" |
| 자동 삭제 완료 | 삭제 후 | "주변 아이템이 삭제되었습니다." |

---

## 알림 전송 함수

```python
from mastodon import Mastodon

mastodon = Mastodon(
    access_token=BOT_ACCESS_TOKEN,
    api_base_url=MASTODON_API_BASE_URL
)

def send_notification(
    recipient_id: str,
    message: str,
    visibility: str = 'direct'
):
    """마스토돈 알림 전송

    Args:
        recipient_id: 수신자 마스토돈 ID
        message: 메시지 내용 (@멘션 포함)
        visibility: 공개 범위 (direct, unlisted, public)
    """
    try:
        mastodon.status_post(
            f"@{recipient_id}\n{message}",
            visibility=visibility
        )
        return True
    except Exception as e:
        print(f"알림 전송 실패: {e}")
        return False
```

## 알림 템플릿

```python
class NotificationTemplates:
    """알림 메시지 템플릿"""

    @staticmethod
    def item_received(sender_name: str, item_name: str, location: str) -> str:
        return (
            f"📦 {sender_name}님이 {item_name}을(를) 양도했습니다!\n"
            f"{location}에서 확인하세요."
        )

    @staticmethod
    def bag_full(item_name: str, volume: int, available: int) -> str:
        return (
            f"⚠️ 가방 공간이 부족합니다!\n"
            f"아이템: {item_name} (부피: {volume})\n"
            f"남은 공간: {available}칸\n\n"
            f"주변에 임시 보관됩니다."
        )

    @staticmethod
    def cleanup_warning(items: list) -> str:
        item_list = ', '.join(item['name'] for item in items[:3])
        if len(items) > 3:
            item_list += f" 외 {len(items) - 3}개"

        return (
            f"⏰ 1시간 후 주변 아이템이 삭제됩니다!\n"
            f"아이템: {item_list}\n"
            f"지금 가방에 넣으세요!"
        )

    @staticmethod
    def cleanup_done(items: list) -> str:
        item_list = '\n'.join(f"- {item['name']} x{item['quantity']}" for item in items)

        return (
            f"⚠️ 주변 아이템이 자동 삭제되었습니다:\n"
            f"{item_list}\n\n"
            f"다음부터는 빨리 가방에 넣으세요!"
        )

    @staticmethod
    def item_used(item_name: str, effect: str, remaining: int) -> str:
        msg = f"✨ {item_name}을(를) 사용했습니다!\n{effect}"
        if remaining > 0:
            msg += f"\n남은 수량: {remaining}개"
        else:
            msg += f"\n(마지막 {item_name})"
        return msg
```

## 알림 설정 (선택)

플레이어별 알림 설정을 관리합니다.

```python
# 알림 설정 (시트에 컬럼 추가 또는 별도 저장)
NOTIFICATION_SETTINGS = {
    'default': {
        'item_received': True,
        'bag_full': True,
        'cleanup_warning': True,
        'cleanup_done': True
    }
}

def should_notify(mastodon_id: str, notification_type: str) -> bool:
    """알림 전송 여부 확인"""
    settings = NOTIFICATION_SETTINGS.get(mastodon_id, NOTIFICATION_SETTINGS['default'])
    return settings.get(notification_type, True)

def send_notification_if_enabled(
    recipient_id: str,
    notification_type: str,
    message: str
):
    """설정에 따라 알림 전송"""
    if should_notify(recipient_id, notification_type):
        send_notification(recipient_id, message)
```
