# 5.1 - 주변 아이템 자동삭제

## 개요

주변에 있는 아이템을 정기적으로 자동 삭제합니다.

---

## 삭제 규칙

- 매일 0시에 실행
- 모든 캐릭터의 '주변' 컬럼을 비움
- 삭제 전 마스토돈으로 알림 전송 (선택)

---

## 크론잡 구현

### schedule 라이브러리

```python
import schedule
import time
from datetime import datetime

def cleanup_nearby_items():
    """주변 아이템 자동 삭제"""
    print(f"[{datetime.now()}] 주변 아이템 자동 삭제 시작...")

    characters = get_all_characters()
    deleted_count = 0

    for char in characters:
        nearby = char.get('주변', '')
        if nearby and nearby.strip():
            # 삭제할 아이템 기록
            items = parse_inventory(nearby)

            # 주변 비우기
            update_cell(char['이름'], '주변', '')
            deleted_count += len(items)

            # 알림 전송 (선택)
            if items:
                notify_deleted_items(char['아이디'], char['이름'], items)

    print(f"[{datetime.now()}] 완료. 삭제된 아이템: {deleted_count}개")

def notify_deleted_items(mastodon_id, char_name, items):
    """삭제 알림 전송"""
    item_list = '\n'.join(f"- {item['name']} x{item['quantity']}" for item in items)

    message = (
        f"@{mastodon_id}\n"
        f"⚠️ {char_name}의 주변 아이템이 자동 삭제되었습니다:\n"
        f"{item_list}\n\n"
        f"다음부터는 빨리 가방에 넣으세요!"
    )

    mastodon.status_post(message, visibility='direct')

# 스케줄 설정
schedule.every().day.at("00:00").do(cleanup_nearby_items)

def run_scheduler():
    """스케줄러 실행"""
    while True:
        schedule.run_pending()
        time.sleep(60)

if __name__ == '__main__':
    run_scheduler()
```

---

## APScheduler 사용 (대안)

```python
from apscheduler.schedulers.background import BackgroundScheduler
from apscheduler.triggers.cron import CronTrigger

scheduler = BackgroundScheduler()

scheduler.add_job(
    cleanup_nearby_items,
    CronTrigger(hour=0, minute=0),  # 매일 0시
    id='cleanup_nearby',
    replace_existing=True
)

scheduler.start()
```

## 봇과 통합

메인 봇 프로세스에 스케줄러를 통합합니다.

```python
import threading

def start_bot():
    """봇 시작 (스케줄러 포함)"""
    # 스케줄러를 별도 스레드로 실행
    scheduler_thread = threading.Thread(target=run_scheduler, daemon=True)
    scheduler_thread.start()

    # 메인 폴링 루프
    main()

if __name__ == '__main__':
    start_bot()
```

## 삭제 전 경고 (선택)

삭제 1시간 전에 경고 알림을 보냅니다.

```python
def warn_before_cleanup():
    """삭제 1시간 전 경고"""
    characters = get_all_characters()

    for char in characters:
        nearby = char.get('주변', '')
        if nearby and nearby.strip():
            items = parse_inventory(nearby)
            if items:
                item_list = ', '.join(f"{item['name']}" for item in items[:3])
                if len(items) > 3:
                    item_list += f" 외 {len(items) - 3}개"

                message = (
                    f"@{char['아이디']}\n"
                    f"⏰ 1시간 후 주변 아이템이 삭제됩니다!\n"
                    f"아이템: {item_list}\n"
                    f"지금 가방에 넣으세요!"
                )
                mastodon.status_post(message, visibility='direct')

# 매일 23시에 경고
schedule.every().day.at("23:00").do(warn_before_cleanup)
```
