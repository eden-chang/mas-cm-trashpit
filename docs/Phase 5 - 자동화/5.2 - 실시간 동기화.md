# 5.2 - 실시간 동기화

## 개요

웹과 봇 간의 데이터 동기화를 구현합니다.

---

## 동기화 시나리오

```
[봇에서 아이템 사용] → [시트 업데이트] → [웹에 반영]

[웹에서 가방 정리] → [시트 업데이트] → [봇 응답에 반영]
```

## 폴링 방식

웹에서 주기적으로 서버를 확인합니다.

### JavaScript (웹)

```javascript
let lastKnownUpdate = null;
let hasUnsavedChanges = false;
const POLL_INTERVAL = 5000; // 5초

async function startPolling() {
    setInterval(async () => {
        if (!currentCharacter) return;

        try {
            const response = await fetch(`/api/character/${currentCharacter}`);
            const data = await response.json();

            // 타임스탬프 비교 (있다면)
            if (data.last_update && data.last_update !== lastKnownUpdate) {
                if (hasUnsavedChanges) {
                    // 저장하지 않은 변경사항 있음
                    showNotification(
                        '⚠️ 다른 곳에서 변경되었습니다. 저장 후 새로고침하세요.',
                        'warning'
                    );
                } else {
                    // 자동 새로고침
                    await loadCharacterData(currentCharacter);
                    showNotification('데이터가 업데이트되었습니다.');
                }
                lastKnownUpdate = data.last_update;
            }
        } catch (error) {
            console.error('Polling error:', error);
        }
    }, POLL_INTERVAL);
}

function markAsUnsaved() {
    hasUnsavedChanges = true;
    document.getElementById('save-btn').classList.add('highlight');
}

function markAsSaved() {
    hasUnsavedChanges = false;
    document.getElementById('save-btn').classList.remove('highlight');
}
```

## 충돌 감지

동시 편집 충돌을 감지합니다.

### 서버 (Flask)

```python
@app.route('/api/bag/<name>', methods=['POST'])
def update_bag(name):
    client_timestamp = request.json.get('last_known_update')
    server_timestamp = get_timestamp(name)

    # 충돌 감지
    if client_timestamp and server_timestamp != client_timestamp:
        return jsonify({
            'success': False,
            'error': '다른 곳에서 변경되었습니다. 새로고침 후 다시 시도하세요.',
            'conflict': True,
            'server_timestamp': server_timestamp
        }), 409

    # 정상 업데이트
    items = request.json.get('items', [])
    update_bag_items(name, items)
    update_timestamp(name)

    return jsonify({
        'success': True,
        'timestamp': get_timestamp(name)
    })
```

### 클라이언트 (JavaScript)

```javascript
async function saveBag() {
    const items = collectBagItems();

    const response = await fetch(`/api/bag/${currentCharacter}`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
            items: items,
            last_known_update: lastKnownUpdate
        })
    });

    const result = await response.json();

    if (result.conflict) {
        // 충돌 발생
        if (confirm('데이터가 변경되었습니다. 새로고침하시겠습니까?')) {
            await loadCharacterData(currentCharacter);
        }
        return;
    }

    if (result.success) {
        lastKnownUpdate = result.timestamp;
        markAsSaved();
        showNotification('저장되었습니다!');
    } else {
        showNotification(result.error, 'error');
    }
}
```
