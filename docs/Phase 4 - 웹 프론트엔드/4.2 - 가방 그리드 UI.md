## JavaScript

```javascript
const BAG_COLUMNS = 10;
const BAG_ROWS = 4;

let bagItems = [];
let gridState = [];

function initBagGrid() {
    const grid = document.getElementById('bag-grid');
    grid.innerHTML = '';

    gridState = Array(BAG_ROWS).fill(null).map(() =>
        Array(BAG_COLUMNS).fill(null)
    );

    for (let row = 0; row < BAG_ROWS; row++) {
        for (let col = 0; col < BAG_COLUMNS; col++) {
            const cell = document.createElement('div');
            cell.className = 'grid-cell';
            cell.dataset.row = row;
            cell.dataset.col = col;

            // 드롭 이벤트
            cell.addEventListener('dragover', handleDragOver);
            cell.addEventListener('drop', handleDrop);
            cell.addEventListener('dragleave', handleDragLeave);

            grid.appendChild(cell);
        }
    }
}

function renderBagItems(items) {
    bagItems = items;
    const list = document.getElementById('bag-items');
    list.innerHTML = '';

    let totalVolume = 0;

    items.forEach(item => {
        const li = document.createElement('li');
        li.draggable = true;
        li.dataset.itemName = item.name;
        li.dataset.volume = item.volume;

        const volume = item.volume * item.quantity;
        totalVolume += volume;

        li.innerHTML = `
            <span>${item.name} x${item.quantity}</span>
            <span class="volume">부피: ${volume}</span>
        `;

        li.addEventListener('dragstart', handleItemDragStart);
        li.addEventListener('dragend', handleItemDragEnd);

        list.appendChild(li);
    });

    // 용량 표시
    document.getElementById('bag-used').textContent = totalVolume;
    updateCapacityDisplay(totalVolume);
}

function updateCapacityDisplay(used) {
    const capacity = parseInt(document.getElementById('bag-capacity').textContent);
    const usedEl = document.getElementById('bag-used');

    if (used > capacity) {
        usedEl.classList.add('over');
    } else {
        usedEl.classList.remove('over');
    }
}

// 그리드에 아이템 시각화
function placeItemsOnGrid(items) {
    // 그리드 초기화
    gridState = Array(BAG_ROWS).fill(null).map(() =>
        Array(BAG_COLUMNS).fill(null)
    );

    document.querySelectorAll('.grid-cell').forEach(cell => {
        cell.classList.remove('occupied');
        cell.textContent = '';
    });

    // 아이템 배치 (단순히 순서대로)
    let currentRow = 0;
    let currentCol = 0;

    items.forEach(item => {
        for (let i = 0; i < item.quantity * item.volume; i++) {
            if (currentCol >= BAG_COLUMNS) {
                currentCol = 0;
                currentRow++;
            }
            if (currentRow >= BAG_ROWS) break;

            const cell = document.querySelector(
                `.grid-cell[data-row="${currentRow}"][data-col="${currentCol}"]`
            );
            if (cell) {
                cell.classList.add('occupied');
                cell.textContent = item.name.charAt(0);
                gridState[currentRow][currentCol] = item.name;
            }
            currentCol++;
        }
    });
}
```

## 용량별 그리드 크기

```javascript
function getGridSize(capacity) {
    // 용량에 따른 그리드 크기
    if (capacity <= 20) {
        return { rows: 2, cols: 10 };  // 20칸
    } else if (capacity <= 40) {
        return { rows: 4, cols: 10 };  // 40칸
    } else {
        return { rows: 6, cols: 10 };  // 60칸
    }
}

function initBagGridByCapacity(capacity) {
    const { rows, cols } = getGridSize(capacity);
    BAG_ROWS = rows;
    BAG_COLUMNS = cols;
    initBagGrid();
}
```
