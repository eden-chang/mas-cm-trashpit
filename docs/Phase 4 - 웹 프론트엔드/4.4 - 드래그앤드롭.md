# 4.4 - 드래그앤드롭

## 개요

주변 아이템을 가방으로 드래그하는 기능을 구현합니다.

---

## 드래그 이벤트 흐름

```
[주변 아이템] ─── dragstart ───► [드래그 중]
                                      │
                                 dragover
                                      │
              ┌───────────────────────┼───────────────────────┐
              │                       │                       │
         [가방 그리드]           [휴지통]              [드래그 취소]
              │                       │                       │
            drop                    drop                  dragend
              │                       │                       │
         [가방에 추가]           [아이템 삭제]          [원위치]
```

---

## 드래그 시작

```javascript
let draggedItem = null;

function handleNearbyDragStart(e) {
    draggedItem = {
        name: e.target.dataset.itemName,
        volume: parseInt(e.target.dataset.volume),
        quantity: 1,
        source: 'nearby'
    };

    e.target.classList.add('dragging');
    e.dataTransfer.effectAllowed = 'move';
    e.dataTransfer.setData('text/plain', JSON.stringify(draggedItem));

    // 드롭 영역 하이라이트
    document.querySelectorAll('.grid-cell').forEach(cell => {
        cell.classList.add('drop-target-hint');
    });
    document.getElementById('trash-zone').classList.add('active');
}

function handleNearbyDragEnd(e) {
    e.target.classList.remove('dragging');
    draggedItem = null;

    // 하이라이트 제거
    document.querySelectorAll('.grid-cell').forEach(cell => {
        cell.classList.remove('drop-target-hint', 'drop-target');
    });
    document.getElementById('trash-zone').classList.remove('active', 'drag-over');
}
```

## 드래그 오버 (가방 그리드)

```javascript
function handleDragOver(e) {
    e.preventDefault();

    if (!draggedItem) return;

    // 공간 확인
    const available = getAvailableSpace();
    if (draggedItem.volume <= available) {
        e.target.classList.add('drop-target');
        e.dataTransfer.dropEffect = 'move';
    } else {
        e.dataTransfer.dropEffect = 'none';
    }
}

function handleDragLeave(e) {
    e.target.classList.remove('drop-target');
}
```

## 드롭 (가방 그리드)

```javascript
async function handleDrop(e) {
    e.preventDefault();
    e.target.classList.remove('drop-target');

    if (!draggedItem) return;

    // 공간 확인
    const available = getAvailableSpace();
    if (draggedItem.volume > available) {
        showNotification('가방 공간이 부족합니다!', 'error');
        return;
    }

    if (draggedItem.source === 'nearby') {
        // 주변 → 가방 이동
        try {
            await moveItemToBag(draggedItem.name);
        } catch (error) {
            showNotification('이동 실패: ' + error.message, 'error');
        }
    }

    draggedItem = null;
}

async function moveItemToBag(itemName) {
    const response = await fetch(`/api/nearby/${currentCharacter}/move`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
            item_name: itemName,
            quantity: 1
        })
    });

    if (!response.ok) {
        const error = await response.json();
        throw new Error(error.error);
    }

    showNotification(`${itemName}을(를) 가방에 넣었습니다!`);
    await loadCharacterData(currentCharacter);
}
```

## 휴지통 드롭

```javascript
function initTrashZone() {
    const trash = document.getElementById('trash-zone');

    trash.addEventListener('dragover', (e) => {
        e.preventDefault();
        if (draggedItem) {
            trash.classList.add('drag-over');
            e.dataTransfer.dropEffect = 'move';
        }
    });

    trash.addEventListener('dragleave', () => {
        trash.classList.remove('drag-over');
    });

    trash.addEventListener('drop', async (e) => {
        e.preventDefault();
        trash.classList.remove('drag-over');

        if (!draggedItem) return;

        // 확인 대화상자
        if (confirm(`${draggedItem.name}을(를) 버리시겠습니까?`)) {
            await discardItem(draggedItem.name, draggedItem.source);
        }

        draggedItem = null;
    });
}

async function discardItem(itemName, source) {
    // 위치에 따라 다른 처리
    const endpoint = source === 'nearby'
        ? `/api/nearby/${currentCharacter}/discard`
        : `/api/bag/${currentCharacter}/discard`;

    const response = await fetch(endpoint, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ item_name: itemName, quantity: 1 })
    });

    if (response.ok) {
        showNotification(`${itemName}을(를) 버렸습니다.`);
        await loadCharacterData(currentCharacter);
    } else {
        const error = await response.json();
        showNotification(error.error, 'error');
    }
}
```

## 알림 표시

```javascript
function showNotification(message, type = 'success') {
    const notification = document.createElement('div');
    notification.className = `notification ${type}`;
    notification.textContent = message;

    document.body.appendChild(notification);

    // 애니메이션
    setTimeout(() => notification.classList.add('show'), 10);
    setTimeout(() => {
        notification.classList.remove('show');
        setTimeout(() => notification.remove(), 300);
    }, 3000);
}
```
