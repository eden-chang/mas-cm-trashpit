### API 모듈 예시

```typescript
// src/lib/api.ts
const BASE = import.meta.env.VITE_API_BASE || '/api';

async function request<T>(endpoint: string, options?: RequestInit): Promise<T> {
  const res = await fetch(`${BASE}${endpoint}`, {
    headers: { 'Content-Type': 'application/json', ...options?.headers },
    ...options,
  });
  const data = await res.json();
  if (!res.ok) throw new Error(data.error || '요청 실패');
  return data as T;
}

export interface CharacterSummary {
  name: string;
  bag_capacity: number;
  bag_used: number;
  nearby_count: number;
}

export interface BagItemUpdate {
  name: string;
  quantity: number;
}

export async function getCharacters(): Promise<CharacterSummary[]> {
  return request('/characters');
}

export async function getCharacter(name: string) {
  return request(`/character/${encodeURIComponent(name)}`);
}

export async function getBag(name: string) {
  return request(`/bag/${encodeURIComponent(name)}`);
}

export async function updateBag(name: string, items: BagItemUpdate[]) {
  return request(`/bag/${encodeURIComponent(name)}`, {
    method: 'POST',
    body: JSON.stringify({ items }),
  });
}

export async function getNearby(name: string) {
  return request(`/nearby/${encodeURIComponent(name)}`);
}

export async function moveToBag(name: string, itemName: string, quantity = 1) {
  return request(`/nearby/${encodeURIComponent(name)}/move`, {
    method: 'POST',
    body: JSON.stringify({ item_name: itemName, quantity }),
  });
}
```
