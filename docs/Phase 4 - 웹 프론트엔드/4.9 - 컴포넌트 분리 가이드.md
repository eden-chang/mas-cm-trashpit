### 3. hooks/useInventory.ts

```typescript
import { useState, useCallback } from 'react';
import type { Item, NearbyItem, FreeItem } from '../types';

interface UseInventoryProps {
  initialItems?: Item[];
  initialNearbyItems?: NearbyItem[];
  initialFreeItems?: FreeItem[];
  capacity?: number;
}

export function useInventory({
  initialItems = [],
  initialNearbyItems = [],
  initialFreeItems = [],
  capacity = 60,
}: UseInventoryProps) {
  const [items, setItems] = useState<Item[]>(initialItems);
  const [nearbyItems, setNearbyItems] = useState<NearbyItem[]>(initialNearbyItems);
  const [freeItems, setFreeItems] = useState<FreeItem[]>(initialFreeItems);
  const [hasUnsavedChanges, setHasUnsavedChanges] = useState(false);

  const usedCapacity = items.reduce((sum, item) => sum + item.volume, 0);
  const remainingCapacity = capacity - usedCapacity;

  const addItem = useCallback((item: Item) => {
    setItems(prev => [...prev, item]);
    setHasUnsavedChanges(true);
  }, []);

  const removeItem = useCallback((itemId: string) => {
    setItems(prev => prev.filter(i => i.id !== itemId));
    setHasUnsavedChanges(true);
  }, []);

  const moveItem = useCallback((itemId: string, newPosition: { row: number; col: number }) => {
    setItems(prev => prev.map(i =>
      i.id === itemId ? { ...i, gridPosition: newPosition } : i
    ));
    setHasUnsavedChanges(true);
  }, []);

  const moveNearbyToBag = useCallback((nearbyId: string, position: { row: number; col: number }) => {
    const nearby = nearbyItems.find(n => n.id === nearbyId);
    if (!nearby) return;

    // Remove from nearby
    setNearbyItems(prev => prev.filter(n => n.id !== nearbyId));

    // Add to bag
    const newItem: Item = {
      id: `item-${Date.now()}`,
      name: nearby.name,
      count: 1,
      volume: nearby.volume,
      gridPosition: position,
      shapeIndex: 0,
    };
    setItems(prev => [...prev, newItem]);
    setHasUnsavedChanges(true);
  }, [nearbyItems]);

  const discardItem = useCallback((itemId: string) => {
    setItems(prev => prev.filter(i => i.id !== itemId));
    setHasUnsavedChanges(true);
  }, []);

  return {
    items,
    nearbyItems,
    freeItems,
    capacity,
    usedCapacity,
    remainingCapacity,
    hasUnsavedChanges,
    setItems,
    setNearbyItems,
    setFreeItems,
    addItem,
    removeItem,
    moveItem,
    moveNearbyToBag,
    discardItem,
    setHasUnsavedChanges,
  };
}
```

### 4. grid/GridCell.tsx

```typescript
import { useDrop } from 'react-dnd';
import type { Item } from '../types';

interface GridCellProps {
  row: number;
  col: number;
  isOccupied: boolean;
  occupiedBy?: Item;
  onDrop: (item: Item, row: number, col: number) => void;
  canDrop: (item: Item, row: number, col: number) => boolean;
}

export function GridCell({ row, col, isOccupied, occupiedBy, onDrop, canDrop }: GridCellProps) {
  const [{ isOver, canDropHere }, drop] = useDrop(() => ({
    accept: ['ITEM', 'NEARBY_ITEM'],
    drop: (item: Item) => onDrop(item, row, col),
    canDrop: (item: Item) => canDrop(item, row, col),
    collect: (monitor) => ({
      isOver: monitor.isOver(),
      canDropHere: monitor.canDrop(),
    }),
  }), [row, col, onDrop, canDrop]);

  return (
    <div
      ref={drop}
      className={`
        w-10 h-10 rounded-md border transition-all duration-200
        ${isOccupied
          ? 'bg-cyan/15 border-cyan/40'
          : 'bg-white/3 border-white/10'}
        ${isOver && canDropHere
          ? 'bg-success/20 border-success/50 border-dashed'
          : ''}
        ${isOver && !canDropHere
          ? 'bg-danger/20 border-danger/50'
          : ''}
      `}
    />
  );
}
```

```typescript
import { useDrop } from 'react-dnd';
import { Trash2 } from 'lucide-react';
import type { Item } from '../types';

interface TrashZoneProps {
  onDiscard: (item: Item) => void;
}

export function TrashZone({ onDiscard }: TrashZoneProps) {
  const [{ isOver }, drop] = useDrop(() => ({
    accept: ['ITEM', 'NEARBY_ITEM'],
    drop: (item: Item) => onDiscard(item),
    collect: (monitor) => ({
      isOver: monitor.isOver(),
    }),
  }), [onDiscard]);

  return (
    <div
      ref={drop}
      className={`
        flex flex-col items-center justify-center gap-2 p-4 rounded-xl
        border-2 border-dashed transition-all duration-200
        ${isOver
          ? 'bg-danger/20 border-danger shadow-glow-danger'
          : 'bg-danger/10 border-danger/30'}

    >
      <Trash2 className={`w-6 h-6 ${isOver ? 'text-danger' : 'text-danger/60'}`} />
      <span className={`text-sm ${isOver ? 'text-danger' : 'text-danger/60'}`}>
        Drop to Delete
      </span>
    </div>
  );
}
```

### 6. dialogs/QuantityDialog.tsx

```typescript
import { useState } from 'react';
import { Dialog, DialogContent, DialogHeader, DialogTitle } from '@/components/ui/dialog';
import { Button } from '@/components/ui/button';

interface QuantityDialogProps {
  open: boolean;
  onClose: () => void;
  itemName: string;
  maxQuantity: number;
  onConfirm: (quantity: number) => void;
}

export function QuantityDialog({
  open,
  onClose,
  itemName,
  maxQuantity,
  onConfirm
}: QuantityDialogProps) {
  const [quantity, setQuantity] = useState(1);

  const handleConfirm = () => {
    onConfirm(quantity);
    setQuantity(1);
    onClose();
  };

  return (
    <Dialog open={open} onOpenChange={onClose}>
      <DialogContent className="bg-bg-overlay border-white/10">
        <DialogHeader>
          <DialogTitle className="text-white">Select Quantity</DialogTitle>
        </DialogHeader>
        <div className="space-y-4">
          <p className="text-white/70">
            How many <span className="text-cyan">{itemName}</span> to move?
          </p>
          <div className="flex items-center gap-4">
            <Button
              variant="outline"
              size="sm"
              onClick={() => setQuantity(Math.max(1, quantity - 1))}
              disabled={quantity <= 1}
            >
              -
            </Button>
            <span className="text-xl text-white font-bold w-12 text-center">
              {quantity}
            </span>
            <Button
              variant="outline"
              size="sm"
              onClick={() => setQuantity(Math.min(maxQuantity, quantity + 1))}
              disabled={quantity >= maxQuantity}
            >
              +
            </Button>
          </div>
          <div className="flex justify-end gap-2">
            <Button variant="ghost" onClick={onClose}>
              Cancel
            </Button>
            <Button onClick={handleConfirm}>
              Confirm
            </Button>
          </div>
        </div>
      </DialogContent>
    </Dialog>
  );
}
```
