```javascript
function renderMiscItems(items) {
    const list = document.getElementById('misc-items');
    list.innerHTML = '';

    items.forEach(item => {
        const li = document.createElement('li');
        li.innerHTML = `
            <span class="icon">📎</span>
            <span class="name">${item.name}</span>
            <span class="count">x${item.quantity}</span>

        list.appendChild(li);
    });

    if (items.length === 0) {
        list.innerHTML = '<li class="empty">아이템이 없습니다.</li>';
    }
}
```

```javascript
function renderNearbyItems(items, availableSpace) {
    const list = document.getElementById('nearby-items');
    list.innerHTML = '';

    let totalVolume = 0;

    items.forEach(item => {
        const li = document.createElement('li');
        li.className = 'nearby-item';
        li.draggable = true;
        li.dataset.itemName = item.name;
        li.dataset.volume = item.volume;

        const volume = item.volume * item.quantity;
        totalVolume += volume;

        const canFit = item.volume <= availableSpace;

        li.innerHTML = `
            <div class="item-info">
                <div class="item-name">📦 ${item.name} x${item.quantity}</div>
                <div class="item-volume">부피: ${volume}</div>
            </div>
            <div class="item-actions">
                <button class="primary move-to-bag"
                        data-item="${item.name}"
                        ${canFit ? '' : 'disabled'}>
                    ${canFit ? '가방에 넣기' : '공간 부족'}
                </button>
            </div>
        `;

        // 드래그 이벤트
        li.addEventListener('dragstart', handleNearbyDragStart);
        li.addEventListener('dragend', handleNearbyDragEnd);

        list.appendChild(li);
    });

    if (items.length === 0) {
        list.innerHTML = '<li class="empty">주변에 아이템이 없습니다.</li>';
    }

    // 요약
    document.getElementById('nearby-volume').textContent = totalVolume;
    document.getElementById('available-space').textContent = availableSpace;
}

// 가방에 넣기 버튼
document.addEventListener('click', async (e) => {
    if (e.target.classList.contains('move-to-bag')) {
        const itemName = e.target.dataset.item;
        await moveItemToBag(itemName);
    }
});

async function moveItemToBag(itemName) {
    const response = await fetch(`/api/nearby/${currentCharacter}/move`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ item_name: itemName, quantity: 1 })
    });

    const result = await response.json();

    if (result.success) {
        showNotification(`${itemName}을(를) 가방에 넣었습니다!`);
        loadCharacterData(currentCharacter);
    } else {
        showNotification(result.error, 'error');
    }
}
```

## 탭 전환

```javascript
function initTabs() {
    document.querySelectorAll('.tab').forEach(tab => {
        tab.addEventListener('click', () => {
            const targetId = tab.dataset.tab;

            // 탭 버튼 활성화
            document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
            tab.classList.add('active');

            // 탭 패널 표시
            document.querySelectorAll('.tab-panel').forEach(panel => {
                panel.style.display = 'none';
            });
            document.getElementById(`${targetId}-tab`).style.display = 'block';
        });
    });
}
```
