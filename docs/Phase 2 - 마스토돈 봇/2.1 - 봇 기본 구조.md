## 마스토돈 연결

```python
from mastodon import Mastodon
import os

# 마스토돈 인스턴스 생성
mastodon = Mastodon(
    access_token=os.getenv('BOT_ACCESS_TOKEN'),
    api_base_url=os.getenv('MASTODON_API_BASE_URL')
)

def reply(status_id: str, message: str):
    """답글 전송 (unlisted 가시성)"""
    mastodon.status_reply(
        to_status=mastodon.status(status_id),
        status=message,
        visibility='unlisted'
    )
```

## 명령어 패턴

```python
import re

# 정규식 패턴 정의
PATTERNS = {
    'acquire': re.compile(r'\[획득/([^\]]+)\]'),
    'discard': re.compile(r'\[버리기/([^\]]+)\]'),
    'use': re.compile(r'\[사용/([^\]]+)\]'),
    'give_item': re.compile(r'\[양도/([^\]/]+)/([^\]]+)\]'),
    'give_point': re.compile(r'\[양도/(\d+)포인트/([^\]]+)\]'),
    'admin_give': re.compile(r'\[지급/([^\]/]+)/([^\]]+)\]'),
}

def parse_command(content: str) -> tuple[str, list] | None:
    """명령어 파싱"""
    # HTML 태그 제거
    content = re.sub(r'<[^>]+>', '', content)

    for cmd_type, pattern in PATTERNS.items():
        match = pattern.search(content)
        if match:
            return (cmd_type, list(match.groups()))

    return None
```

## 명령어 핸들러

```python
def on_notification(notification):
    """멘션 알림 처리"""
    if notification['type'] != 'mention':
        return

    status = notification['status']
    content = status['content']
    user = notification['account']['acct']  # 마스토돈 ID
    status_id = status['id']

    # 명령어 파싱
    result = parse_command(content)
    if not result:
        return

    cmd_type, args = result

    # 캐릭터 조회 (마스토돈 ID로)
    character = get_character_by_mastodon_id(user)
    if not character:
        reply(status_id, f"@{user} 등록된 캐릭터를 찾을 수 없습니다.")
        return

    # 명령어 실행
    handlers = {
        'acquire': handle_acquire,
        'discard': handle_discard,
        'use': handle_use,
        'give_item': handle_give_item,
        'give_point': handle_give_point,
        'admin_give': handle_give_admin,
    }

    handler = handlers.get(cmd_type)
    if handler:
        handler(status_id, user, character, args)
```

## 구글 시트 연동

### 캐릭터 조회

```python
import gspread
from oauth2client.service_account import ServiceAccountCredentials

# 구글 시트 연결
scope = ['https://spreadsheets.google.com/feeds']
creds = ServiceAccountCredentials.from_json_keyfile_name('credentials.json', scope)

def get_character_by_mastodon_id(mastodon_id: str) -> dict | None:
    """마스토돈 ID로 캐릭터 조회"""
    management_ws = sheet.worksheet('관리')
    records = management_ws.get_all_records()[1:]  # 2행 스킵

    for record in records:
        if record['아이디'] == mastodon_id:
            return record

    return None

def get_character_by_name(name: str) -> dict | None:
    """캐릭터 이름으로 조회"""
    management_ws = sheet.worksheet('관리')
    records = management_ws.get_all_records()[1:]

    for record in records:
        if record['이름'] == name:
            return record

    return None
```

### 인벤토리 파싱

```python
def parse_inventory(inventory_str: str) -> list[dict]:
    """인벤토리 문자열 파싱
    
    형식: "아이템명: 수량, 아이템명: 수량"
    반환: [{'name': '아이템명', 'quantity': 수량}, ...]
    """
    if not inventory_str:
        return []
    
    items = []
    for part in inventory_str.split(','):
        part = part.strip()
        if ':' in part:
            name, qty = part.split(':', 1)
            items.append({
                'name': name.strip(),
                'quantity': int(qty.strip())
            })
    
    return items

def find_character_row(char_name: str) -> int | None:
    """캐릭터 이름으로 시트 행 번호 찾기"""
    management_ws = sheet.worksheet('관리')
    records = management_ws.get_all_records()
    
    for idx, record in enumerate(records, start=2):  # 헤더가 1행
        if record['이름'] == char_name:
            return idx
    
    return None

def get_column_number(column_name: str) -> int:
    """컬럼 이름으로 컬럼 번호 반환"""
    column_map = {
        '이름': 1,
        '아이디': 2,
        '주변': 3,
        '가방': 4,
        '여유공간': 5,
        '가방부피': 6,
        '체력': 7,
        '근력': 8,
        '행운': 9,
        '포인트': 10,
    }
    return column_map.get(column_name, 0)
```

## 폴링 루프

```python
import time

def main():
    """메인 폴링 루프"""
    last_id = None

    while True:
        try:
            notifications = mastodon.notifications(
                types=['mention'],
                since_id=last_id
            )

            for notif in reversed(notifications):
                on_notification(notif)
                last_id = notif['id']

        except Exception as e:
            print(f"Error: {e}")

        time.sleep(POLLING_INTERVAL)

if __name__ == '__main__':
    main()
```
