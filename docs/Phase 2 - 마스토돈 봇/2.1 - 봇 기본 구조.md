## 마스토돈 연결

```python
from mastodon import Mastodon
import os

# 마스토돈 인스턴스 생성
mastodon = Mastodon(
    access_token=os.getenv('BOT_ACCESS_TOKEN'),
    api_base_url=os.getenv('MASTODON_API_BASE_URL')
)

def reply(status_id: str, message: str):
    """답글 전송 (unlisted 가시성)"""
    mastodon.status_reply(
        to_status=mastodon.status(status_id),
        status=message,
        visibility='unlisted'
    )
```

## 명령어 패턴

```python
import re

# 정규식 패턴 정의
PATTERNS = {
    'acquire': re.compile(r'\[획득/([^\]]+)\]'),
    'discard': re.compile(r'\[버리기/([^\]]+)\]'),
    'use': re.compile(r'\[사용/([^\]]+)\]'),
    'give_item': re.compile(r'\[양도/([^\]/]+)/([^\]]+)\]'),
    'give_point': re.compile(r'\[양도/(\d+)포인트/([^\]]+)\]'),
    'admin_give': re.compile(r'\[지급/([^\]/]+)/([^\]]+)\]'),
}

def parse_command(content: str) -> tuple[str, list] | None:
    """명령어 파싱"""
    # HTML 태그 제거
    content = re.sub(r'<[^>]+>', '', content)

    for cmd_type, pattern in PATTERNS.items():
        match = pattern.search(content)
        if match:
            return (cmd_type, list(match.groups()))

    return None
```

## 명령어 핸들러

```python
def on_notification(notification):
    """멘션 알림 처리"""
    if notification['type'] != 'mention':
        return

    status = notification['status']
    content = status['content']
    user = notification['account']['acct']  # 마스토돈 ID
    status_id = status['id']

    # 명령어 파싱
    result = parse_command(content)
    if not result:
        return

    cmd_type, args = result

    # 캐릭터 조회 (마스토돈 ID로)
    character = get_character_by_mastodon_id(user)
    if not character:
        reply(status_id, f"@{user} 등록된 캐릭터를 찾을 수 없습니다.")
        return

    # 명령어 실행
    handlers = {
        'acquire': handle_acquire,
        'discard': handle_discard,
        'use': handle_use,
        'give_item': handle_give_item,
        'give_point': handle_give_point,
        'admin_give': handle_give_admin,
    }

    handler = handlers.get(cmd_type)
    if handler:
        handler(status_id, user, character, args)
```

### 캐릭터 조회

```python
from shared.supabase_client import get_supabase, TABLE_CHARACTERS
from shared.parser import parse_json_inventory

def get_character_by_mastodon_id(mastodon_id: str) -> dict | None:
    """마스토돈 ID로 캐릭터 조회"""
    supabase = get_supabase()
    response = (
        supabase.table(TABLE_CHARACTERS)
        .select("*")
        .eq("id", mastodon_id)
        .limit(1)
        .execute()
    )
    
    if not response.data:
        return None
    
    row = response.data[0]
    return {
        'name': row.get('name', ''),
```

### 인벤토리 파싱

```python
def parse_json_inventory(data) -> list[dict]:
    """JSON 인벤토리 파싱
    
    형식: {"아이템명": 수량, ...}
    반환: [{'name': '아이템명', 'quantity': 수량}, ...]
    """
    if not data or not isinstance(data, dict):
        return []
    
    return [
        {'name': name, 'quantity': qty}
        for name, qty in data.items()
        if qty > 0
    ]

def serialize_json_inventory(items: list[dict]) -> dict:
    """아이템 리스트를 JSON으로 직렬화
    
    형식: [{'name': '아이템명', 'quantity': 수량}, ...]
    반환: {"아이템명": 수량, ...}
    """
    result = {}
    for item in items:
        name = item.get('name', '')
        qty = item.get('quantity', 0)
        if name and qty > 0:
            result[name] = result.get(name, 0) + qty
    return result
```

## 폴링 루프

```python
import time

def main():
    """메인 폴링 루프"""
    last_id = None

    while True:
        try:
            notifications = mastodon.notifications(
                types=['mention'],
                since_id=last_id
            )

            for notif in reversed(notifications):
                on_notification(notif)
                last_id = notif['id']

        except Exception as e:
            print(f"Error: {e}")

        time.sleep(POLLING_INTERVAL)

if __name__ == '__main__':
    main()
```
