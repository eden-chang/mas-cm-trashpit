# 2.3 - 사용 명령어

## 개요

`[사용/아이템명]` 명령어로 아이템을 사용합니다.

---

## 처리 흐름

```
1. 아이템 위치 확인 (주변 → 가방 → 여유공간)
2. 아이템 정보 조회 (상점)
3. 사용 가능 여부 확인
4. 효과 적용 (스탯 변경)
5. 아이템 수량 차감
6. 응답 전송 (사용문구 포함)
```

---

## 구현

```python
def handle_use(status_id: str, user: str, character: dict, args: list):
    """[사용/아이템명] 처리"""
    item_name = args[0].strip()
    char_name = character['이름']

    # 1. 아이템 위치 확인
    location = find_item_location(char_name, item_name)

    if not location:
        reply(status_id, f"@{user} '{item_name}'을(를) 소지하고 있지 않습니다.")
        return

    # 2. 아이템 정보 조회
    item_info = get_item_info(item_name)

    if not item_info:
        reply(status_id, f"@{user} '{item_name}' 정보를 찾을 수 없습니다.")
        return

    # 3. 사용 가능 여부 확인
    if not is_usable(item_info):
        reply(status_id, f"@{user} '{item_name}'은(는) 사용할 수 없는 아이템입니다.")
        return

    # 4. 효과 적용 (다이스 굴림 포함)
    actual_value = apply_effect(char_name, item_info)

    # 5. 아이템 차감
    if location == 'nearby':
        remove_from_nearby(char_name, item_name, 1)
    elif location == 'bag':
        remove_from_bag(char_name, item_name, 1)
    elif location == 'misc':
        remove_from_misc(char_name, item_name, 1)

    # 6. 응답
    use_message = item_info['use_message']
    effect_text = format_effect(item_info['stat'], actual_value)
    remaining = get_item_count(char_name, item_name)

    response = f"@{user} {use_message}\n{effect_text}"
    if remaining > 0:
        response += f"\n남은 {item_name}: {remaining}개"
    else:
        response += f"\n(마지막 {item_name}을(를) 사용했습니다)"

    reply(status_id, response)
```

## 효과 적용

```python
import re
import random

def roll_dice(dice_str: str) -> int:
    """다이스 표기법 처리

    지원 형식: 정수, ndm, ndm+k, ndm-k, -(ndm+k)
    예: '10', '1d10', '1d6+3', '1d6-2', '-(1d6+3)'
    """
    if not dice_str or dice_str.strip() == '':
        return 0

    dice_str = str(dice_str).strip()

    # 정수인 경우
    try:
        return int(dice_str)
    except ValueError:
        pass

    # 음수 다이스: -(ndm+k) 형태
    negative = False
    if dice_str.startswith('-(') and dice_str.endswith(')'):
        negative = True
        dice_str = dice_str[2:-1]  # 괄호 제거
    elif dice_str.startswith('-'):
        negative = True
        dice_str = dice_str[1:]

    # ndm+k 또는 ndm-k 파싱
    match = re.match(r'(\d+)d(\d+)([+-]\d+)?', dice_str)
    if match:
        n = int(match.group(1))  # 주사위 개수
        m = int(match.group(2))  # 주사위 면수
        k = int(match.group(3) or 0)  # 보정값

        # 주사위 굴리기
        total = sum(random.randint(1, m) for _ in range(n)) + k
        return -total if negative else total

    return 0

def apply_effect(char_name: str, item_info: dict) -> int:
    """아이템 효과 적용 (스탯 변경). 실제 적용된 수치 반환"""
    stat = item_info.get('stat', '')
    value_str = item_info.get('value', '')

    if not stat or stat == '사용 불가' or not value_str:
        return 0  # 효과 없음

    # 스탯 컬럼 매핑
    stat_columns = {
        '체력': '체력',
        '근력': '근력',
        '행운': '행운',
    }

    column = stat_columns.get(stat)
    if not column:
        return 0

    # 다이스 굴리기
    value = roll_dice(value_str)

    # 현재 값 가져오기
    management_ws = sheet.worksheet('관리')
    row_num = find_character_row(char_name)
    col_num = get_column_number(column)

    current = int(management_ws.cell(row_num, col_num).value or 0)
    new_value = current + value

    # 업데이트
    management_ws.update_cell(row_num, col_num, new_value)

    return value

def format_effect(stat: str, value: int) -> str:
    """효과 텍스트 포맷"""
    if not stat or stat == '사용 불가' or value == 0:
        return ""

    sign = '+' if value > 0 else ''
    return f"효과: {stat} {sign}{value}"
```

## 아이템 수량 확인

```python
def get_item_count(char_name: str, item_name: str) -> int:
    """캐릭터가 소지한 아이템 총 수량"""
    char = get_character_by_name(char_name)
    total = 0

    for col in ['주변', '가방', '여유공간']:
        items = parse_inventory(char[col])
        for item in items:
            if item['name'] == item_name:
                total += item['quantity']

    return total
```

## 응답 예시

### 정상 사용

```
@발트 사과를 먹었다! 상큼한 맛이 난다.
효과: 체력 +3
남은 사과: 2개
```

### 마지막 아이템 사용

```
@발트 의료키트를 사용했다! 상처가 치료되었다.
효과: 체력 +10
(마지막 의료키트를 사용했습니다)
```

### 사용 불가

```
@발트 '칼'은(는) 사용할 수 없는 아이템입니다.
```

### 미소지

```
@발트 '물약'을(를) 소지하고 있지 않습니다.
```
