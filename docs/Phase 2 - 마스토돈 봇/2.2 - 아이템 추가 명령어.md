# 2.2 - 획득 및 버리기 명령어

## 개요

플레이어가 아이템을 획득하거나 버리는 명령어를 처리합니다.

---

## [획득/아이템명]

플레이어가 아이템을 획득합니다.

### 처리 흐름

```
1. 상점에서 아이템 정보 확인
2. 부피 확인
   - 부피 0 → 여유공간에 추가
   - 부피 1+ → 주변에 추가
3. 시트 업데이트
4. 응답 전송
```

### 구현

```python
def handle_acquire(status_id: str, user: str, character: dict, args: list):
    """[획득/아이템명] 처리"""
    item_name = args[0].strip()
    char_name = character['이름']

    # 아이템 정보 확인
    item_info = get_item_info(item_name)
    if not item_info:
        reply(status_id, f"@{user} '{item_name}'은(는) 존재하지 않는 아이템입니다.")
        return

    volume = item_info['volume']

    if volume == 0:
        # 여유공간에 추가
        add_to_misc(char_name, item_name, 1)
        reply(status_id,
            f"@{user} {item_name}을(를) 획득했습니다! (부피 0)\n"
            f"자동으로 여유공간에 보관되었습니다.")
    else:
        # 주변에 추가
        add_to_nearby(char_name, item_name, 1)

        # 가방 공간 확인
        available = get_available_space(char_name)

        if volume <= available:
            reply(status_id,
                f"@{user} {item_name}을(를) 획득했습니다! (부피: {volume})\n"
                f"가방 남은 공간: {available}칸\n"
                f"웹에서 가방에 넣으세요.")
        else:
            reply(status_id,
                f"@{user} {item_name}을(를) 획득했습니다! (부피: {volume})\n"
                f"⚠️ 가방 공간이 부족합니다! (남은 공간: {available}칸)\n"
                f"주변에 임시 보관됩니다. (자동 삭제 주의)")
```

---

## [버리기/아이템명]

플레이어가 아이템을 버립니다.

### 처리 흐름

```
1. 아이템 위치 확인 (주변 → 가방 → 여유공간)
2. 해당 위치에서 삭제
3. 시트 업데이트
4. 응답 전송
```

### 구현

```python
def handle_discard(status_id: str, user: str, character: dict, args: list):
    """[버리기/아이템명] 처리"""
    item_name = args[0].strip()
    char_name = character['이름']

    # 아이템 위치 찾기
    location = find_item_location(char_name, item_name)

    if not location:
        reply(status_id, f"@{user} '{item_name}'을(를) 소지하고 있지 않습니다.")
        return

    # 삭제 처리
    if location == 'nearby':
        remove_from_nearby(char_name, item_name, 1)
    elif location == 'bag':
        remove_from_bag(char_name, item_name, 1)
    elif location == 'misc':
        remove_from_misc(char_name, item_name, 1)

    location_name = {'nearby': '주변', 'bag': '가방', 'misc': '여유공간'}
    reply(status_id, f"@{user} {item_name}을(를) 버렸습니다. ({location_name[location]}에서)")
```

## 유틸리티 함수

```python
def find_item_location(char_name: str, item_name: str) -> str | None:
    """아이템 위치 찾기 (주변 → 가방 → 여유공간 순)"""
    char = get_character_by_name(char_name)

    # 주변 확인
    nearby = parse_inventory(char['주변'])
    if any(item['name'] == item_name for item in nearby):
        return 'nearby'

    # 가방 확인
    bag = parse_inventory(char['가방'])
    if any(item['name'] == item_name for item in bag):
        return 'bag'

    # 여유공간 확인
    misc = parse_inventory(char['여유공간'])
    if any(item['name'] == item_name for item in misc):
        return 'misc'

    return None

def add_to_nearby(char_name: str, item_name: str, quantity: int):
    """주변에 아이템 추가"""
    update_inventory_column(char_name, '주변', item_name, quantity)

def add_to_misc(char_name: str, item_name: str, quantity: int):
    """여유공간에 아이템 추가"""
    update_inventory_column(char_name, '여유공간', item_name, quantity)

def remove_from_nearby(char_name: str, item_name: str, quantity: int):
    """주변에서 아이템 제거"""
    update_inventory_column(char_name, '주변', item_name, -quantity)

def remove_from_bag(char_name: str, item_name: str, quantity: int):
    """가방에서 아이템 제거"""
    update_inventory_column(char_name, '가방', item_name, -quantity)

def remove_from_misc(char_name: str, item_name: str, quantity: int):
    """여유공간에서 아이템 제거"""
    update_inventory_column(char_name, '여유공간', item_name, -quantity)
```

## 응답 예시

### 획득 성공 (부피 0)

```
@발트 사과를 획득했습니다! (부피 0)
자동으로 여유공간에 보관되었습니다.
```

### 획득 성공 (부피 1+, 공간 충분)

```
@발트 로프를 획득했습니다! (부피: 2)
가방 남은 공간: 10칸
웹에서 가방에 넣으세요.
```

### 획득 성공 (공간 부족)

```
@발트 텐트를 획득했습니다! (부피: 5)
⚠️ 가방 공간이 부족합니다! (남은 공간: 2칸)
주변에 임시 보관됩니다. (자동 삭제 주의)
```

### 버리기 성공

```
@발트 돌멩이를 버렸습니다. (주변에서)
```

### 에러: 존재하지 않는 아이템

```
@발트 '이상한물건'은(는) 존재하지 않는 아이템입니다.
```

### 에러: 미소지 아이템

```
@발트 '칼'을(를) 소지하고 있지 않습니다.
```

## 공통 유틸리티 함수

### 인벤토리 업데이트

```python
def update_inventory_column(char_name: str, column: str, item_name: str, delta: int):
    """인벤토리 컬럼 업데이트 (delta: 양수=추가, 음수=제거)"""
    management_ws = sheet.worksheet('관리')

    # 캐릭터 행 찾기
    row_num = find_character_row(char_name)
    if not row_num:
        return

    # 현재 인벤토리 가져오기
    col_num = get_column_number(column)
    current = management_ws.cell(row_num, col_num).value or ''

    # 파싱 및 업데이트
    items = parse_inventory(current)
    updated = False

    for item in items:
        if item['name'] == item_name:
            item['quantity'] += delta
            updated = True
            break

    if not updated and delta > 0:
        items.append({'name': item_name, 'quantity': delta})

    # 수량 0 이하 제거
    items = [item for item in items if item['quantity'] > 0]

    # 문자열로 변환 (형식: "아이템명: 수량, 아이템명: 수량")
    new_value = ', '.join(f"{item['name']}: {item['quantity']}" for item in items)

    # 시트 업데이트
    management_ws.update_cell(row_num, col_num, new_value)
```

### 아이템 마스터 조회

```python
def get_item_info(item_name: str) -> dict | None:
    """상점 시트에서 아이템 정보 조회"""
    shop_ws = sheet.worksheet('상점')
    records = shop_ws.get_all_records()
    
    for record in records:
        if record['이름'] == item_name:
            return {
                'name': record['이름'],
                'volume': int(record['부피']),
                'stat': record.get('스탯', ''),
                'value': record.get('수치', ''),
                'use_message': record.get('사용문구', ''),
            }
    
    return None

def get_available_space(char_name: str) -> int:
    """가방 남은 공간 계산"""
    char = get_character_by_name(char_name)
    max_capacity = int(char.get('가방부피', 20))
    
    # 현재 가방 아이템 부피 합산
    bag_items = parse_inventory(char['가방'])
    used = 0
    
    for item in bag_items:
        item_info = get_item_info(item['name'])
        if item_info:
            used += item_info['volume'] * item['quantity']
    
    return max_capacity - used
```
