# 2.4 - 양도 명령어

## 개요

다른 플레이어에게 아이템이나 포인트를 전달합니다.

---

## [양도/아이템명/이름]

아이템을 다른 캐릭터에게 양도합니다.

### 처리 흐름

```
1. 수신자 캐릭터 확인
2. 아이템 위치 확인 (발신자)
3. 아이템 정보 조회
4. 발신자에서 제거
5. 수신자 주변(또는 여유공간)에 추가
6. 양쪽에 알림
```

### 구현

```python
def handle_give_item(status_id: str, user: str, character: dict, args: list):
    """[양도/아이템명/이름] 처리"""
    item_name = args[0].strip()
    recipient_name = args[1].strip()
    sender_name = character['이름']

    # 1. 수신자 확인
    recipient = get_character_by_name(recipient_name)
    if not recipient:
        reply(status_id, f"@{user} '{recipient_name}' 캐릭터를 찾을 수 없습니다.")
        return

    # 자기 자신에게 양도 방지
    if sender_name == recipient_name:
        reply(status_id, f"@{user} 자기 자신에게는 양도할 수 없습니다.")
        return

    # 2. 아이템 위치 확인
    location = find_item_location(sender_name, item_name)
    if not location:
        reply(status_id, f"@{user} '{item_name}'을(를) 소지하고 있지 않습니다.")
        return

    # 3. 아이템 정보
    item_info = get_item_info(item_name)
    if not item_info:
        reply(status_id, f"@{user} '{item_name}' 정보를 찾을 수 없습니다.")
        return

    # 4. 발신자에서 제거
    if location == 'nearby':
        remove_from_nearby(sender_name, item_name, 1)
    elif location == 'bag':
        remove_from_bag(sender_name, item_name, 1)
    elif location == 'misc':
        remove_from_misc(sender_name, item_name, 1)

    # 5. 수신자에게 추가
    if item_info['volume'] == 0:
        add_to_misc(recipient_name, item_name, 1)
        dest = '여유공간'
    else:
        add_to_nearby(recipient_name, item_name, 1)
        dest = '주변'

    # 6. 응답
    reply(status_id, f"@{user} {item_name}을(를) {recipient_name}에게 양도했습니다!")

    # 수신자에게도 알림 (수신자 마스토돈 ID로)
    recipient_id = recipient['아이디']
    notify_recipient(recipient_id, sender_name, item_name, dest)

def notify_recipient(recipient_id: str, sender_name: str, item_name: str, dest: str):
    """수신자에게 DM 또는 멘션"""
    message = (
        f"@{recipient_id} {sender_name}님이 {item_name}을(를) 양도했습니다!\n"
        f"{dest}에서 확인하세요."
    )
    mastodon.status_post(message, visibility='direct')
```

## [양도/n포인트/이름]

포인트를 다른 캐릭터에게 양도합니다.

### 처리 흐름

```
1. 수신자 캐릭터 확인
2. 발신자 포인트 확인
3. 포인트 차감/추가
4. 양쪽에 알림
```

### 구현

```python
def handle_give_point(status_id: str, user: str, character: dict, args: list):
    """[양도/n포인트/이름] 처리"""
    amount = int(args[0])
    recipient_name = args[1].strip()
    sender_name = character['이름']

    # 유효성 검사
    if amount <= 0:
        reply(status_id, f"@{user} 양도할 포인트는 1 이상이어야 합니다.")
        return

    # 수신자 확인
    recipient = get_character_by_name(recipient_name)
    if not recipient:
        reply(status_id, f"@{user} '{recipient_name}' 캐릭터를 찾을 수 없습니다.")
        return

    if sender_name == recipient_name:
        reply(status_id, f"@{user} 자기 자신에게는 양도할 수 없습니다.")
        return

    # 포인트 확인 (포인트 컬럼이 있다고 가정)
    sender_points = get_character_points(sender_name)
    if sender_points < amount:
        reply(status_id, f"@{user} 포인트가 부족합니다. (보유: {sender_points})")
        return

    # 포인트 이동
    update_points(sender_name, -amount)
    update_points(recipient_name, amount)

    # 응답
    remaining = sender_points - amount
    reply(status_id,
        f"@{user} {amount}포인트를 {recipient_name}에게 양도했습니다!\n"
        f"남은 포인트: {remaining}")

    # 수신자 알림
    recipient_id = recipient['아이디']
    mastodon.status_post(
        f"@{recipient_id} {sender_name}님이 {amount}포인트를 양도했습니다!",
        visibility='direct'
    )
```

## 포인트 유틸리티

```python
def get_character_points(char_name: str) -> int:
    """캐릭터 포인트 조회"""
    # 포인트 컬럼이 있다면
    char = get_character_by_name(char_name)
    return int(char.get('포인트', 0))

def update_points(char_name: str, delta: int):
    """포인트 업데이트"""
    management_ws = sheet.worksheet('관리')
    row_num = find_character_row(char_name)
    col_num = get_column_number('포인트')

    current = int(management_ws.cell(row_num, col_num).value or 0)
    management_ws.update_cell(row_num, col_num, current + delta)
```
