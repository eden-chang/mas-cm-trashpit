# 3.2 - Supabase 연동

## 개요

Supabase PostgreSQL과 연동하는 모듈을 구현합니다.

---

## 인증 설정

### 환경 변수

`.env` 파일에 Supabase 연결 정보를 설정합니다.

```env
SUPABASE_URL=https://your-project.supabase.co
SUPABASE_SERVICE_KEY=your_service_role_key
```

**주의**: `SUPABASE_SERVICE_KEY`는 Service Role 키를 사용합니다. 이 키는 RLS(Row Level Security)를 우회하므로 서버 측에서만 사용해야 합니다.

## Supabase 클라이언트 (싱글톤)

```python
# shared/supabase_client.py
from typing import Optional
from supabase import create_client, Client
from . import config

_client: Optional[Client] = None

def get_supabase() -> Client:
    """Supabase 클라이언트 반환 (싱글톤)"""
    global _client
    
    if _client is None:
        _client = create_client(
            config.SUPABASE_URL,
            config.SUPABASE_SERVICE_KEY
        )
    
    return _client

# 테이블 이름 상수
TABLE_CHARACTERS = "characters"
TABLE_ITEMS = "items"
```

## 데이터 읽기

### 캐릭터 조회

```python
from shared.supabase_client import get_supabase, TABLE_CHARACTERS

def get_all_characters() -> list[dict]:
    """모든 캐릭터 조회"""
    supabase = get_supabase()
    response = supabase.table(TABLE_CHARACTERS).select("*").execute()
    return response.data

def get_character_by_name(name: str) -> dict | None:
    """이름으로 캐릭터 조회 (PK 조회)"""
    supabase = get_supabase()
    response = (
        supabase.table(TABLE_CHARACTERS)
        .select("*")
        .eq("name", name)
        .limit(1)
        .execute()
    )
    return response.data[0] if response.data else None

def get_character_by_mastodon_id(mastodon_id: str) -> dict | None:
    """마스토돈 ID로 캐릭터 조회"""
    supabase = get_supabase()
    response = (
        supabase.table(TABLE_CHARACTERS)
        .select("*")
        .eq("id", mastodon_id)
        .limit(1)
        .execute()
    )
    return response.data[0] if response.data else None
```

```python
from shared.supabase_client import get_supabase, TABLE_ITEMS

def get_all_items() -> list[dict]:
    """모든 아이템 조회"""
    supabase = get_supabase()
    response = supabase.table(TABLE_ITEMS).select("*").execute()
    return response.data

def get_item_info(item_name: str) -> dict | None:
    """아이템 정보 조회 (PK 조회)"""
    supabase = get_supabase()
    response = (
        supabase.table(TABLE_ITEMS)
        .select("*")
        .eq("name", item_name)
        .limit(1)
        .execute()
    )
    
    if not response.data:
        return None
    
    item = response.data[0]
    return {
        'name': item['name'],
        'price': item['price'],
        'description': item['description'],
        'use_message': item['use_script'],
        'stat': item['change_stats'],
        'value': item['change_value'],
        'volume': item['size']
    }
```

```python
def update_bag_items(name: str, items: dict) -> bool:
    """가방 아이템 업데이트
    
    Args:
        name: 캐릭터 이름
        items: {"아이템명": 수량, ...} 형식의 딕셔너리
```

### 복합 업데이트

```python
def update_inventory(
    name: str,
    bag_items: dict,
    nearby_items: dict,
    misc_items: dict,
    bag_layout: list[dict] | None = None
) -> bool:
    """가방·주변·여유공간 한 번에 업데이트"""
    supabase = get_supabase()
    
    update_data = {
        "bag": bag_items,
        "misc": misc_items,
        "around": nearby_items,
    }
    
    if bag_layout is not None:
        update_data["arrange"] = bag_layout
    
    response = (
        supabase.table(TABLE_CHARACTERS)
        .update(update_data)
        .eq("name", name)
        .execute()
    )
    return bool(response.data)
```

## JSON 인벤토리 파싱

```python
def parse_json_inventory(data) -> list[dict]:
    """JSON 인벤토리를 아이템 리스트로 변환
    
    Args:
        data: {"아이템명": 수량, ...} 또는 None
        
    Returns:
        [{"name": "아이템명", "quantity": 수량}, ...]
    """
    if not data:
        return []
    
    if isinstance(data, dict):
        return [
            {"name": name, "quantity": qty}
            for name, qty in data.items()
            if qty > 0
        ]
    
    return []

def serialize_json_inventory(items: list[dict]) -> dict:
    """아이템 리스트를 JSON 인벤토리로 직렬화
    
    Args:
        items: [{"name": "아이템명", "quantity": 수량}, ...]
        
    Returns:
        {"아이템명": 수량, ...}
    """
    result = {}
    for item in items:
        name = item.get("name", "")
        qty = item.get("quantity", 0)
        if name and qty > 0:
            result[name] = result.get(name, 0) + qty
    return result
```

## 에러 처리

```python
from postgrest.exceptions import APIError
import logging

logger = logging.getLogger(__name__)

def safe_supabase_operation(func, *args, retries=3, **kwargs):
    """재시도 로직이 포함된 Supabase 작업"""
    import time
    
    for attempt in range(retries):
        try:
            return func(*args, **kwargs)
        except APIError as e:
            logger.warning("Supabase API 오류 (시도 %d/%d): %s", attempt + 1, retries, e)
            if attempt < retries - 1:
                time.sleep(2 ** attempt)
            else:
                raise
        except Exception as e:
            logger.exception("Supabase 작업 실패: %s", e)
            raise
    
    raise Exception("최대 재시도 횟수 초과")
```
