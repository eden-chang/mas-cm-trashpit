# 3.2 - 시트 연동

## 개요

구글 시트와 연동하는 모듈을 구현합니다.

---

## 인증 설정

### credentials.json

Google Cloud Console에서 서비스 계정을 생성하고 JSON 키를 다운로드합니다.

```json
{
  "type": "service_account",
  "project_id": "your-project-id",
  "private_key_id": "...",
  "private_key": "-----BEGIN PRIVATE KEY-----\n...\n-----END PRIVATE KEY-----\n",
  "client_email": "bot@your-project.iam.gserviceaccount.com",
  "client_id": "...",
  "auth_uri": "https://accounts.google.com/o/oauth2/auth",
  "token_uri": "https://oauth2.googleapis.com/token"
}
```

**주의**: 시트 공유 설정에서 `client_email`에 편집 권한을 부여해야 합니다.

## 시트 클라이언트

```python
import gspread
from oauth2client.service_account import ServiceAccountCredentials
import os

SHEET_ID = os.getenv('SHEET_ID')
CREDENTIALS_PATH = os.getenv('GOOGLE_CREDENTIALS_PATH', 'credentials.json')

def get_sheet_client():
    """구글 시트 클라이언트 생성"""
    scope = [
        'https://spreadsheets.google.com/feeds',
        'https://www.googleapis.com/auth/drive'
    ]
    creds = ServiceAccountCredentials.from_json_keyfile_name(CREDENTIALS_PATH, scope)
    client = gspread.authorize(creds)
    return client.open_by_key(SHEET_ID)

# 싱글톤
_sheet = None

def get_sheet():
    global _sheet
    if _sheet is None:
        _sheet = get_sheet_client()
    return _sheet
```

## 행 번호 찾기

```python
def find_character_row(char_name: str) -> int | None:
    """캐릭터 이름으로 행 번호 찾기 (1-indexed)"""
    ws = get_management_ws()
    names = ws.col_values(get_column_number('이름'))

    for i, name in enumerate(names):
        if name == char_name:
            return i + 1  # 1-indexed

    return None

def get_column_number(column_name: str) -> int:
    """컬럼 이름으로 컬럼 번호 반환 (1-indexed)"""
    columns = {
        '이름': 1,
        '아이디': 2,
        '진영': 3,
        '체력': 4,
        '근력': 5,
        '행운': 6,
        '소지금': 7,
        '가방': 8,
        '여유공간': 9,
        '주변': 10,
    }
    return columns.get(column_name, 1)
```

## 데이터 읽기

```python
def get_all_characters() -> list[dict]:
    """모든 캐릭터 조회"""
    ws = get_management_ws()
    records = ws.get_all_records()
    return records[1:]  # 2행(정보행) 스킵

def get_character_by_name(name: str) -> dict | None:
    """이름으로 캐릭터 조회"""
    for char in get_all_characters():
        if char['이름'] == name:
            return char
    return None

def get_character_by_mastodon_id(mastodon_id: str) -> dict | None:
    """마스토돈 ID로 캐릭터 조회"""
    for char in get_all_characters():
        if char['아이디'] == mastodon_id:
            return char
    return None

def get_all_items() -> list[dict]:
    """모든 아이템 조회"""
    ws = get_shop_ws()
    records = ws.get_all_records()
    return records[1:]  # 2행 스킵

def get_item_info(item_name: str) -> dict | None:
    """아이템 정보 조회"""
    for item in get_all_items():
        if item['아이템명'] == item_name:
            return {
                'name': item['아이템명'],
                'price': item['가격'],
                'description': item['설명'],
                'use_message': item['사용문구'],
                'stat': item['스탯'],
                'value': item['수치'],
                'volume': item['부피']
            }
    return None
```

## 데이터 쓰기

```python
def update_cell(char_name: str, column: str, value):
    """특정 셀 업데이트"""
    ws = get_management_ws()
    row = find_character_row(char_name)
    col = get_column_number(column)

    if row:
        ws.update_cell(row, col, value)

def update_inventory(char_name: str, column: str, items: list[dict]):
    """인벤토리 컬럼 업데이트"""
    # items: [{'name': '사과', 'quantity': 3}, ...]
    value = ','.join(f"{item['name']}:{item['quantity']}" for item in items if item['quantity'] > 0)
    update_cell(char_name, column, value)

def update_bag_items(char_name: str, items: list[dict]):
    """가방 아이템 업데이트"""
    update_inventory(char_name, '가방', items)

def update_nearby_items(char_name: str, items: list[dict]):
    """주변 아이템 업데이트"""
    update_inventory(char_name, '주변', items)

def update_misc_items(char_name: str, items: list[dict]):
    """여유공간 아이템 업데이트"""
    update_inventory(char_name, '여유공간', items)
```

## 에러 처리

```python
from gspread.exceptions import APIError
import time

def safe_update_cell(char_name: str, column: str, value, retries=3):
    """재시도 로직이 포함된 셀 업데이트"""
    for attempt in range(retries):
        try:
            update_cell(char_name, column, value)
            return True
        except APIError as e:
            if e.response.status_code == 429:  # Rate limit
                time.sleep(2 ** attempt)
            else:
                raise
    return False
```
