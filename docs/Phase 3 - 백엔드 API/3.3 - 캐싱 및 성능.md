## 메모리 캐시

### 단순 캐시

```python
import time
from functools import wraps

CACHE_TTL = int(os.getenv('CACHE_TTL', 3600))  # 1시간

_cache = {}

def cached(ttl=CACHE_TTL):
    """TTL 기반 캐시 데코레이터"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            key = f"{func.__name__}:{args}:{kwargs}"

            if key in _cache:
                value, timestamp = _cache[key]
                if time.time() - timestamp < ttl:
                    return value

            result = func(*args, **kwargs)
            _cache[key] = (result, time.time())
            return result

        return wrapper
    return decorator

def invalidate_cache(pattern=None):
    """캐시 무효화"""
    global _cache
    if pattern:
        _cache = {k: v for k, v in _cache.items() if pattern not in k}
    else:
        _cache = {}
```

### 적용 예시

## Redis 캐시 (선택)

대규모 운영 시 Redis를 사용할 수 있습니다.

```python
import redis
import json

redis_client = redis.Redis(host='localhost', port=6379, db=0)

def redis_cached(ttl=CACHE_TTL):
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            key = f"inv:{func.__name__}:{args}:{kwargs}"

            cached = redis_client.get(key)
            if cached:
                return json.loads(cached)

            result = func(*args, **kwargs)
            redis_client.setex(key, ttl, json.dumps(result))
            return result

        return wrapper
    return decorator

def redis_invalidate(pattern: str):
    """패턴 매칭 캐시 삭제"""
    for key in redis_client.scan_iter(f"inv:*{pattern}*"):
        redis_client.delete(key)
```

## 배치 읽기

여러 데이터를 한 번에 읽어 API 호출을 줄입니다.

```python
def preload_all_data():
    """모든 데이터를 한 번에 로드"""
    global _cache

    # 캐릭터 전체 로드
    characters = get_all_characters()
    for char in characters:
        key = f"get_character_by_name:('{char['이름']}')"
        _cache[key] = (char, time.time())

        key = f"get_character_by_mastodon_id:('{char['아이디']}')"
        _cache[key] = (char, time.time())

    # 아이템 전체 로드
    items = get_all_items()
    for item in items:
        info = {
            'name': item['아이템명'],
            'price': item['가격'],
            'description': item['설명'],
            'use_message': item['사용문구'],
            'stat': item['스탯'],
            'value': item['수치'],
            'volume': item['부피']
        }
        key = f"get_item_info:('{item['아이템명']}')"
        _cache[key] = (info, time.time())

# 시작 시 호출
preload_all_data()
```

### 배치 업데이트

```python
def batch_update_cells(updates: list[tuple]):
    """여러 셀을 한 번에 업데이트

    updates: [(row, col, value), ...]
    """
    ws = get_management_ws()
    cells = []

    for row, col, value in updates:
        cell = ws.cell(row, col)
        cell.value = value
        cells.append(cell)

    ws.update_cells(cells)
```
